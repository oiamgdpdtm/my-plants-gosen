  <script>
    const climateData = {
      1: "冬本番。窓際からのコールドドラフトに注意。夜間は部屋の中央へ。水やりは控えめに。",

      4: "新芽の季節。室内での水やり頻度を少し増やす。",
      5: "生育旺盛期。土の表面が乾いたらたっぷり水やり。",
      6: "梅雨時期。風通しを良くして蒸れを防ぐ。",
      7: "夏の猛暑。エアコンの風が直接当たらない場所に配置。",
      8: "厳暑。締め切った部屋の高湿度と高温に注意。",
      9: "残暑から秋へ。夏の疲れが出やすい時期。",
      10: "秋深まる。最低気温が下がってくるため防寒準備。",
      11: "初冬の冷え込み。五泉市の冬に備え、水やりを控えめに。",
      12: "本格的な冬。エアコン暖房による乾燥を防ぐため葉水を行う。"
    };
    const initialPlants = [
      { id: 1, name: "ガジュマル", tag: "多肉・観葉", waterFreq: "7〜10日に1回", lastWatered: "2026-11-01", memo: "耐寒性高め。乾燥に強いが冬の過湿に注意。", img: "" },
      { id: 2, name: "エバーフレッシュ", tag: "繊細・夜閉じる", waterFreq: "4〜6日に1回", lastWatered: "2026-11-01", memo: "水切れで葉が落ちる。夜間は葉を閉じる。", img: "" },
      { id: 3, name: "サンスベリア", tag: "乾燥重視", waterFreq: "冬はほぼ断水", lastWatered: "2026-10-25", memo: "寒さに弱い。冬場は完全に乾かして温かい場所へ。", img: "" },
      { id: 4, name: "ポトス", tag: "つる性・強健", waterFreq: "5〜7日に1回", lastWatered: "2026-11-01", memo: "耐陰性あり。伸びすぎた葉はカットして水挿しも可能。", img: "" },
      { id: 5, name: "ベンガレンシス", tag: "ゴムの木・人気", waterFreq: "7日に1回", lastWatered: "2026-10-28", memo: "ホコリを定期的に拭き取る。", img: "" }
    ];
    let plants = JSON.parse(localStorage.getItem('my_gosen_plants')) || initialPlants;
    function savePlants() { localStorage.setItem('my_gosen_plants', JSON.stringify(plants)); }
    function updateClimateAdvice() {
      document.getElementById('climateAdviceText').innerText = climateData[document.getElementById('monthSelect').value];
    }
    function renderPlants() {
      const grid = document.getElementById('plantGrid');
      grid.innerHTML = '';
      plants.forEach(p => {
        const card = document.createElement('div');
        card.className = 'plant-card';
        card.innerHTML = `
          ${p.img ? `<img src="${p.img}" class="plant-img" alt="${p.name}">` : `<div class="plant-img">📷 写真未設定</div>`}
          <div class="plant-content">
            <div class="plant-name">${p.name}</div>
            <span class="plant-tag">${p.tag}</span>
            <div class="info-row" style="margin-top:8px;"><span>水やり目安:</span><span class="info-value">${p.waterFreq}</span></div>
            <div class="info-row"><span>前回水やり:</span><span class="info-value">${p.lastWatered || '未記録'}</span></div>
            <div class="memo-box">💡 ${p.memo}</div>
            <div class="btn-group">
              <button class="btn-water" onclick="waterPlant(${p.id})">💧 今日水やりした</button>
              <button class="btn-photo" onclick="document.getElementById('file-${p.id}').click()">📷 写真を変更</button>
              <input type="file" id="file-${p.id}" accept="image/*" onchange="uploadPhoto(event, ${p.id})">
            </div>
          </div>
        `;
        grid.appendChild(card);
      });
    }
    function waterPlant(id) {
      const today = new Date().toISOString().split('T')[0];
      const plant = plants.find(p => p.id === id);
      if (plant) { plant.lastWatered = today; savePlants(); renderPlants(); }
    }
    function uploadPhoto(event, id) {
      const file = event.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(e) {
          const plant = plants.find(p => p.id === id);
          if (plant) { plant.img = e.target.result; savePlants(); renderPlants(); }
        };
        reader.readAsDataURL(file);
      }
    }
    updateClimateAdvice();
    renderPlants();
  </script>
